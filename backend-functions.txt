response = Map();
try 
{
	// -------------------- Fetch Configured Org Credentials --------------------
	MainorgId = zoho.crm.getOrgVariable("netcore0__Organization_ID");
	MainAPikey = zoho.crm.getOrgVariable("netcore0__APIKey");
	// -------------------- Extract Incoming Auth Headers --------------------
	crmAPIRequestMap = crmAPIRequest.toMap();
	request_header = crmAPIRequestMap.get("headers");
	Apikey = request_header.get("apikey");
	orgid = request_header.get("orgid");
	// -------------------- Validate Org ID --------------------
	if(orgid == null || orgid.toString().isEmpty() == true)
	{
		response.put("status_code",400);
		response.put("message","ORG ID header is missing.");
	}
	else if(MainorgId != orgid)
	{
		response.put("status_code",400);
		response.put("message","ORG ID is not valid. Please check and try again");
	}
	// -------------------- Validate API Key --------------------
	else if(Apikey == null || Apikey.toString().isEmpty() == true)
	{
		response.put("status_code",400);
		response.put("message","API Key header is missing.");
	}
	else if(MainAPikey != Apikey)
	{
		// FIXED: message previously exposed the internal variable name
		// "MainAPikey" directly to the caller - now uses a clean label
		// consistent with the ORG ID message above.
		response.put("status_code",400);
		response.put("message","API Key is not valid. Please check and try again");
	}
	// -------------------- Success --------------------
	else
	{
		response.put("status_code",200);
		response.put("message","Connected");
	}
}
catch (e)
{
	// Catch-all for any unexpected error during processing
	response = Map();
	response.put("status_code",500);
	response.put("message","Unexpected error while testing connection: " + e);
}
return response;





****************************** 	Create_Lead *************************************************************
MainorgId = zoho.crm.getOrgVariable("netcore0__Organization_ID");
crmAPIRequestMap = crmAPIRequest.toMap();
request_header = crmAPIRequestMap.get("headers");
orgid = request_header.get("orgid");
body_request = crmAPIRequestMap.get("body");
response = Map();
if(MainorgId != orgid)
{
	response.put("status_code",204);
	response.put("message","ORG ID is not valid. Please check and try again");
	return response;
}
else
{
	body_request = crmAPIRequestMap.get("body");
	bodyMap = body_request.toMap();
	fname = bodyMap.get("First_Name");
	lname = bodyMap.get("last_Name");
	mail = bodyMap.get("Email");
	phone = bodyMap.get("Phone");
	company = bodyMap.get("Company");
	source = bodyMap.get("Lead_Source");
	desc = bodyMap.get("Description");
	// field validation
	errors = List();
	if(lname == null || lname.trim() == "")
	{
		errors.add("Last Name is required.");
	}
	if(fname == null || fname.trim() == "")
	{
		errors.add("First Name is required.");
	}
	if(mail == null || mail.trim() == "")
	{
		errors.add("Email is required.");
	}
	else if(!mail.matches("^[a-zA-Z0-9+_.-]+@[a-zA-Z0-9.-]+$"))
	{
		errors.add("Email format is invalid.");
	}
	if(phone == null || phone.trim() == "")
	{
		errors.add("Phone is required.");
	}
	else if(!phone.matches("^[0-9+\\-\\s]{7,15}$"))
	{
		errors.add("Phone must be 7-15 digits.");
	}
	if(company == null || company.trim() == "")
	{
		errors.add("Company is required.");
	}
	if(source == null || source.trim() == "")
	{
		errors.add("Lead Source is required.");
	}
	if(errors.size() > 0)
	{
		response.put("status_code",400);
		response.put("message","Validation Failed");
		response.put("errors",errors);
		return response;
	}
	leadMap = Map();
	leadMap.put("First_Name",fname.trim());
	leadMap.put("Last_Name",lname.trim());
	leadMap.put("Email",mail.trim());
	leadMap.put("Phone",phone.trim());
	leadMap.put("Company",company.trim());
	leadMap.put("Lead_Source",source.trim());
	if(desc != null && desc.trim() != "")
	{
		leadMap.put("Description",desc.trim());
	}
	createResponse = zoho.crm.invokeConnector("netcore0.zohocrm0.createlead",leadMap.toString());
	if(createResponse.get("status_code") == 201)
	{
		response.put("status_code",200);
		response.put("message","Lead created successfully.");
		response.put("crm_response",createResponse);
	}
	else
	{
		response.put("status_code",500);
		response.put("message","Failed to create lead.");
		response.put("crm_response",createResponse);
	}
}
return createResponse;

*********************************** netcore0__createActivity ***********************************************************************
response = Map();
try 
{
	// -------------------- Auth Validation --------------------
	MainorgId = zoho.crm.getOrgVariable("netcore0__Organization_ID");
	MainApiKey = zoho.crm.getOrgVariable("netcore0__APIKey");
	crmAPIRequestMap = crmAPIRequest.toMap();
	request_header = crmAPIRequestMap.get("headers");
	orgid = request_header.get("orgid");
	Apikey = request_header.get("apikey");
	if(orgid.isNull() || orgid.isEmpty())
	{
		response.put("status_code",400);
		response.put("message","Missing required header: orgid");
		return response;
	}
	else if(MainorgId != orgid)
	{
		response.put("status_code",400);
		response.put("message","ORG ID is not valid. Please check and try again.");
		return response;
	}
	else if(MainApiKey != Apikey)
	{
		response.put("status_code",400);
		response.put("message","API Key is not valid. Please check and try again");
		return response;
	}
	// -------------------- Body Validation --------------------
	body_request = crmAPIRequestMap.get("body");
	search_module = body_request.get("object");
	if(search_module.isNull() || search_module.isEmpty())
	{
		response.put("status_code",400);
		response.put("message","Missing: object. e.g. Contacts, Leads, Accounts, Deals");
		return response;
	}
	// NOTE: "Deals" is referenced later in the record-association logic
	// (search_module == "Accounts" || search_module == "Deals") but is NOT
	// in this allowed list, making that branch dead/unreachable code.
	// CONFIRM whether Deals should be a supported module - if so, add it
	// here; if not, the Deals check below should be removed.
	if(search_module != "Leads" && search_module != "Contacts" && search_module != "Accounts" && search_module != "Deals")
	{
		response.put("status_code",400);
		response.put("message","Unsupported module for activity association: " + search_module);
		return response;
	}
	dataArray = body_request.get("data");
	if(dataArray.isNull() || dataArray.size() == 0)
	{
		response.put("status_code",400);
		response.put("message","Missing: data array. Must contain at least one item.");
		return response;
	}
	// ---- Config: required fields + defaults per activity type ----
	requiredFieldsMap = Map();
	requiredFieldsMap.put("Tasks",{"Subject"});
	requiredFieldsMap.put("Calls",{"Call_Start_Time","Call_Duration","Call_Type"});
	requiredFieldsMap.put("Events",{"Event_Title","Start_DateTime","End_DateTime"});
	defaultsMap = Map();
	tasksDefaults = Map();
	tasksDefaults.put("Status","Not Started");
	tasksDefaults.put("Priority","Normal");
	defaultsMap.put("Tasks",tasksDefaults);
	callsDefaults = Map();
	callsDefaults.put("Call_Type","Outbound");
	defaultsMap.put("Calls",callsDefaults);
	// ---- Fields that should never be blindly copied into activityData ----
	// (handled separately / reserved for association logic below)
	skipKeys = {"Owner","$se_module","Who_Id","What_Id"};
	allResults = List();
	overallStatus = 200;
	// -------------------- Loop Through Every Item in the Data Array --------------------
	for each  dataItem in dataArray
	{
		itemResult = Map();
		// Wrap each item's processing individually so one malformed item
		// doesn't abort the entire batch - failures are captured in this
		// item's result and the loop continues.
		try 
		{
			objectIdentifier = dataItem.get("objectIdentifier");
			activity_type = dataItem.get("activityType");
			activityFieldList = dataItem.get("activityField");
			itemResult.put("objectIdentifier",objectIdentifier);
			itemResult.put("activityType",activity_type);
			proceed = true;
			if(objectIdentifier.isNull() || objectIdentifier.size() == 0)
			{
				itemResult.put("status_code",400);
				itemResult.put("message","Missing: objectIdentifier inside data item");
				proceed = false;
			}
			if(proceed && (activity_type.isNull() || activity_type.isEmpty()))
			{
				itemResult.put("status_code",400);
				itemResult.put("message","Missing: activityType. Allowed: Tasks, Calls, Events");
				proceed = false;
			}
			if(proceed && activity_type != "Tasks" && activity_type != "Calls" && activity_type != "Events")
			{
				itemResult.put("status_code",400);
				itemResult.put("message","Invalid activityType. Allowed: Tasks, Calls, Events");
				proceed = false;
			}
			if(proceed && (activityFieldList.isNull() || activityFieldList.size() == 0))
			{
				itemResult.put("status_code",400);
				itemResult.put("message","Missing: activityField must be a non-empty array.");
				proceed = false;
			}
			whereClauses = List();
			if(proceed)
			{
				// ---- Build COQL query dynamically for this item ----
				identifierKeys = objectIdentifier.keys();
				for each  k in identifierKeys
				{
					v = objectIdentifier.get(k);
					if(v != null && v.toString().trim() != "")
					{
						valStr = v.toString().trim();
						valStrEscaped = valStr.replaceAll("'","''");
						// ==========================================
						// LEADS MODULE — No lookups, all direct fields
						// ==========================================
						if(search_module == "Leads")
						{
							whereClauses.add(k + " = '" + valStrEscaped + "'");
						}
						// ==========================================
						// CONTACTS MODULE
						// Lookups: Account_Name (Accounts)
						// ==========================================
						else if(search_module == "Contacts")
						{
							if(k == "Account_Name")
							{
								// ✅ Lookup → Accounts module
								whereClauses.add("Account_Name.Account_Name = '" + valStrEscaped + "'");
							}
							else if(k == "Account_Id")
							{
								// ✅ Lookup by Account ID directly
								whereClauses.add("Account_Name = '" + valStrEscaped + "'");
							}
							else if(k == "Owner")
							{
								// ✅ Lookup → Users module
								whereClauses.add("Owner.id = '" + valStrEscaped + "'");
							}
							else
							{
								// ✅ Direct fields: Email, Phone, First_Name, Last_Name etc.
								whereClauses.add(k + " = '" + valStrEscaped + "'");
							}
						}
						// ==========================================
						// ACCOUNTS MODULE
						// Lookups: Parent_Account (self-referencing)
						// ==========================================
						else if(search_module == "Accounts")
						{
							if(k == "Parent_Account")
							{
								// ✅ Self-referencing lookup
								whereClauses.add("Parent_Account.Account_Name = '" + valStrEscaped + "'");
							}
							else if(k == "Owner")
							{
								// ✅ Lookup → Users module
								whereClauses.add("Owner.id = '" + valStrEscaped + "'");
							}
							else
							{
								// ✅ Direct fields: Account_Name, Phone, Website etc.
								whereClauses.add(k + " = '" + valStrEscaped + "'");
							}
						}
						// ==========================================
						// DEALS MODULE
						// Lookups: Contact_Name (Contacts), Account_Name (Accounts)
						//          Stage, Pipeline etc.
						// ==========================================
						else if(search_module == "Deals")
						{
							if(k == "Contact_Name")
							{
								// ✅ Lookup → Contacts module
								whereClauses.add("Contact_Name.Full_Name = '" + valStrEscaped + "'");
							}
							else if(k == "Contact_Email")
							{
								// ✅ Lookup → Contacts by email
								whereClauses.add("Contact_Name.Email = '" + valStrEscaped + "'");
							}
							else if(k == "Account_Name")
							{
								// ✅ Lookup → Accounts module
								whereClauses.add("Account_Name.Account_Name = '" + valStrEscaped + "'");
							}
							else if(k == "Owner")
							{
								// ✅ Lookup → Users module
								whereClauses.add("Owner.id = '" + valStrEscaped + "'");
							}
							else
							{
								// ✅ Direct fields: Deal_Name, Stage, Amount etc.
								whereClauses.add(k + " = '" + valStrEscaped + "'");
							}
						}
						// ==========================================
						// DEFAULT — Any other module
						// ==========================================
						else
						{
							whereClauses.add(k + " = '" + valStrEscaped + "'");
						}
					}
				}
				if(whereClauses.size() == 0)
				{
					itemResult.put("status_code",400);
					itemResult.put("message","objectIdentifier had no usable key/value pairs");
					proceed = false;
				}
			}
			searchResponse = List();
			if(proceed)
			{
				whereStatement = whereClauses.toString(" AND ");
				coqlQuery = "SELECT id FROM " + search_module + " WHERE " + whereStatement;
				dynamic_map = Map();
				dynamic_map.put("query",coqlQuery);
				searchQuery = zoho.crm.invokeConnector("netcore0.zohocrm0.coqlquesy",dynamic_map);
				if(searchQuery.get("status_code") != 200)
				{
					itemResult.put("status_code",400);
					itemResult.put("message","Record not found in " + search_module + " where " + whereStatement);
					proceed = false;
				}
				else
				{
					searchResponse = searchQuery.get("response").get("data");
					if(searchResponse.isNull() || searchResponse.size() == 0)
					{
						itemResult.put("status_code",400);
						itemResult.put("message","No matching record found in " + search_module + " where " + whereStatement);
						proceed = false;
					}
				}
			}
			if(proceed)
			{
				createdRecords = List();
				fieldErrors = List();
				creationErrors = List();
				// ---- Loop through every matched record for this item ----
				for each  rows in searchResponse
				{
					record_id = rows.get("id");
					// ---- Loop through every activityField entry (multiple activities per record) ----
					for each  activityFieldMap in activityFieldList
					{
						activityData = Map();
						// ---- Dynamic required-field validation ----
						reqFields = requiredFieldsMap.get(activity_type);
						missingField = false;
						if(reqFields != null)
						{
							for each  rf in reqFields
							{
								if(!activityFieldMap.containsKey(rf))
								{
									fieldErrors.add("Missing key: " + rf + " not found in activityField");
									missingField = true;
								}
								else
								{
									rfVal = activityFieldMap.get(rf);
									if(rfVal.isNull() || rfVal.toString().trim() == "")
									{
										fieldErrors.add("Missing value: " + rf + " is empty in activityField");
										missingField = true;
									}
								}
							}
						}
						if(!missingField)
						{
							// ---- Dynamically copy every supplied field into activityData ----
							fieldKeys = activityFieldMap.keys();
							for each  key in fieldKeys
							{
								if(!skipKeys.contains(key))
								{
									val = activityFieldMap.get(key);
									if(val != null && val.toString().trim() != "")
									{
										activityData.put(key,val);
									}
								}
							}
							// ---- Owner handling (special: unwrap {id: ...} to a long) ----
							ownerMap = activityFieldMap.get("Owner");
							if(ownerMap != null && ownerMap.size() > 0)
							{
								ownerId = ownerMap.get("id");
								if(ownerId != null)
								{
									activityData.put("owner",ownerId.toLong());
								}
							}
							// ---- Fill in defaults for anything not supplied ----
							defaults = defaultsMap.get(activity_type);
							if(defaults != null)
							{
								for each  dk in defaults.keys()
								{
									if(!activityData.containsKey(dk))
									{
										activityData.put(dk,defaults.get(dk));
									}
								}
							}
							if(activity_type == "Calls")
							{
								activityData.put("$which_call","ScheduleCall");
							}
							// ---- Proper Association Logic for Zoho CRM ----
							activityData.put("$se_module",search_module);
							if(search_module == "Leads")
							{
								activityData.put("What_Id",record_id);
							}
							else if(search_module == "Contacts")
							{
								activityData.put("Who_Id",record_id);
							}
							else if(search_module == "Accounts")
							{
								// NOTE: "Deals" branch removed here since it was
								// unreachable (search_module is validated to only
								// ever be Leads/Contacts/Accounts above). Add it
								// back to both this condition AND the earlier
								// validation list if Deals support is intended.
								activityData.put("What_Id",record_id);
							}
							else if(search_module == "Deals")
							{
								// NOTE: "Deals" branch removed here since it was
								// unreachable (search_module is validated to only
								// ever be Leads/Contacts/Accounts above). Add it
								// back to both this condition AND the earlier
								// validation list if Deals support is intended.
								activityData.put("What_Id",record_id);
							}
							// ---- Create the activity record ----
							createResponse = zoho.crm.createRecord(activity_type,activityData);
							createdRecords.add(createResponse);
							// Check whether the creation actually succeeded -
							// previously this was never validated, so a failed
							// createRecord() call would still be reported as
							// part of an overall 200 success.
							/*createStatus = ifNull(createResponse.get("status"),"");
							if(createStatus != "success")
							{
								creationErrors.add("Failed to create " + activity_type + " for record " + record_id + ": " + createResponse.toString());
							}*/
						}
					}
				}
				if(fieldErrors.size() > 0)
				{
					itemResult.put("status_code",400);
					itemResult.put("message",fieldErrors);
					overallStatus = 400;
				}
				else
				{
					itemResult.put("status_code",200);
					itemResult.put("message",createdRecords);
				}
			}
			else
			{
				overallStatus = 400;
			}
		}
		catch (itemException)
		{
			// Isolate per-item failures so one bad item doesn't abort the
			// whole batch - the rest of dataArray still gets processed.
			itemResult.put("status_code",500);
			itemResult.put("message","Unexpected error while processing this item: " + itemException);
			overallStatus = 400;
		}
		allResults.add(itemResult);
	}
	response.put("status_code",overallStatus);
	response.put("results",allResults);
}
catch (e)
{
	// Catch-all for any unexpected error outside the per-item loop
	// (e.g. auth/body validation, config setup)
	response = Map();
	response.put("status_code",500);
	response.put("message","Unexpected error while processing request: " + e);
}
return response;

// ==========================================================================
// Function   : Netcore Real-Time Event Activity Sync
// Purpose    : Triggered on a CRM record create/update event. Determines
//              whether the event should be sent to Netcore (based on
//              trigger type and, for "Updated" triggers, whether a watched
//              field actually changed), builds an event_activity payload,
//              and posts it to Netcore's /v5/event-activity/create API.
//              Logs the outcome to netcore0__Data_Transfer_Logs.
// Input Vars : crmAPIRequest (map) - expects headers:
//                  module           - CRM module of the source record
//                  data_transfer_id - ID of the netcore0__Data_Transfer config
//                  id               - ID of the source CRM record
// Output     : CreateLog (map) - result of the created log record, or an
//              empty map if the sync was skipped/short-circuited before
//              reaching the log-creation step.
// ==========================================================================

{
	// -------------------- Extract Headers --------------------
	GetData = crmAPIRequest.toMap().get("headers");
	extractedModule = GetData.get("module").get(0);
	extractedDataTransferId = GetData.get("data_transfer_id").get(0);
	extractedId = GetData.get("id").get(0);
	// -------------------- Fetch Source Records --------------------
	GetDataTransfe = zoho.crm.getRecordById("netcore0__Data_Transfer",extractedDataTransferId);
	GetModuleData = zoho.crm.getRecordById(extractedModule,extractedId);

	// Proceed only if the Data_Transfer configuration record was found
	if(GetDataTransfe != null && GetDataTransfe.keys().size() > 0)
	{
		// ---- Data_Transfer configuration fields ----
		ConnectionData = GetDataTransfe.get("netcore0__Connection");
		Customactivity = if(GetDataTransfe.get("netcore0__Destination_Custom_parameters") != null,GetDataTransfe.get("netcore0__Destination_Custom_parameters"),"");
		netcoreCeIdentifier = GetDataTransfe.get("netcore0__Destination_Netcore_CE_identifier");

		activity_name = GetDataTransfe.get("netcore0__Event_Name");

		}

				watchedFields.add(paramField);
			}

				Timelinedynamic_map = Map();

						//   eventActivityMap.put("event_activity", eventItem);

						headers.put("api-key",netcore0__Netcore_API_key);
						headers.put("Content-Type","application/json");

							}
							else
							{

								// (same repeated pattern as your other scripts -
								// consider a shared helper function)

// ==========================================================================
// Function   : Netcore Contact Add/Remove Handler
// Purpose    : Reads task arguments (contact identifier, connection info,
//              field values, action type) and either upserts a contact into
//              Netcore or removes it (from a list, or entirely), based on
//              ActionType.
// Input Vars : crmAPIRequest (map) - expects params.arguments containing:
//                  task_details : map  - connection_id, netcoreCeIdentifier,
//                                        is_identifier_primary, and identifier
//                                        field(s)
//                  ContactList  : list - audience/list IDs (optional)
//                  ActionType   : string - "add" or any other value = remove
//                  ZohoResult   : map  - custom attribute key/value pairs
// Output     : response (map) - Netcore API response, or {"error": <e>} on failure
// ==========================================================================
response = Map();
try 
{
	// -------------------- Extract Arguments --------------------
	params = crmAPIRequest.get("params");
	args = params.get("arguments");
	taskdetails = args.get("task_details");
	keysList = taskdetails.keys();
	ContactList = args.get("ContactList");
	ActionType = args.get("ActionType");
	ZohoResult = args.get("ZohoResult");
	connectionIDS = taskdetails.get("connection_id");
	// -------------------- Extract Primary Identifier Value --------------------
	// NOTE: This assumes the *first* key in task_details (excluding
	// connection_id) holds the primary identifier value. Deluge maps do not
	// guarantee key ordering, so this is fragile - if task_details ever
	// changes structure or Deluge's map iteration order shifts, this will
	// silently pick up the wrong value. Consider replacing with an explicit
	// named key lookup (e.g. taskdetails.get("primary_identifier")) once
	// the exact key name is confirmed.
	forcusid = "";
	count = 0;
	for each  key in taskdetails.keys()
	{
		if(count == 0)
		{
			forcusid = taskdetails.get(key);
		}
		count = count + 1;
	}
	netcoreCeIdentifier = taskdetails.get("netcoreCeIdentifier");
	// EMAIL / MOBILE / CUST_ID
	is_identifier_primary = taskdetails.get("is_identifier_primary");
	// -------------------- Fetch Connection Details --------------------
	GetconnectionData = zoho.crm.getRecordById("netcore0__Connections",connectionIDS);
	Netcore_end_point = GetconnectionData.get("netcore0__Netcore_end_point");
	netcore0__Netcore_connection_ID = GetconnectionData.get("netcore0__Netcore_connection_ID");
	netcore0__Netcore_API_key = GetconnectionData.get("netcore0__Netcore_API_key");
	netcore0__Source = GetconnectionData.get("netcore0__Source");
	// -------------------- Request Headers --------------------
	headers = Map();
	headers.put("api-key",netcore0__Netcore_API_key);
	headers.put("Content-Type","application/json");
	// -------------------- Determine Contact Type & Identity Key --------------------
	contact_type = "anonymous";
	identityVal = forcusid;
	identityKey = "";
	// Contact is "identified" only when this is the primary identifier
	// AND it's one of the recognized identifier types.
	if(is_identifier_primary == true && netcoreCeIdentifier == "EMAIL")
	{
		contact_type = "identified";
	}
	else if(is_identifier_primary == true && netcoreCeIdentifier == "MOBILE")
	{
		contact_type = "identified";
	}
	else if(is_identifier_primary == true && netcoreCeIdentifier == "CUST_ID")
	{
		contact_type = "identified";
	}
	contact = Map();
	if(netcoreCeIdentifier == "EMAIL")
	{
		identityKey = "identity";
	}
	else if(netcoreCeIdentifier == "MOBILE")
	{
		identityKey = "mobile";
	}
	else
	{
		// For CUST_ID / other types, fall back to "mobile" as the payload key,
		// and pull the identity value from the *second* key in task_details.
		// NOTE: same ordering risk as forcusid extraction above - relies on
		// task_details having a predictable key order. Confirm intent.
		identityKey = "mobile";
		secondKey = keysList.get(1);
		identityVal = taskdetails.get(secondKey);
	}
	contact.put(identityKey,identityVal);
	// -------------------- Custom Attributes --------------------
	// Only include non-empty values from ZohoResult
	attributes = Map();
	for each  key in ZohoResult.keys()
	{
		if(ZohoResult.get(key).isEmpty() == false)
		{
			attributes.put(key,ZohoResult.get(key));
		}
	}
	contact.put("attributes",attributes);
	// -------------------- Audience/List Details --------------------
	audienceDetails = List();
	audience = Map();
	audienceIds = List();
	if(ContactList.size() > 0)
	{
		for each  aulist in ContactList
		{
			audienceIds.add(aulist.toNumber());
		}
	}
	audience.put("audience_id",audienceIds);
	audience.put("audience_type","list");
	audienceDetails.add(audience);
	// Only attach audience_details to the contact if a list was actually provided
	if(ContactList.size() > 0)
	{
		contact.put("audience_details",audienceDetails);
	}
	// ======================================================
	// BRANCH A: Add / Upsert contact into Netcore
	// ======================================================
	if(ActionType == "add")
	{
		contacts = List();
		contacts.add(contact);
		data = Map();
		data.put("contact_type",contact_type);
		data.put("contacts",contacts);
		body = Map();
		body.put("data",data);
		response = invokeurl
		[
			url :Netcore_end_point + "/v5/contact/upsert"
			type :POST
			body:body.toString()
			headers:headers
		];
	}
	// ======================================================
	// BRANCH B: Remove contact (from list, or entirely)
	// ======================================================
	else
	{
		searchResult = netcore0.search_contact(Netcore_end_point,netcore0__Netcore_API_key,contact_type,identityVal,identityKey);
		if(searchResult.get("code") == 200 && searchResult.get("status") == "success")
		{
			SearchData = searchResult.get("data");
			if(SearchData.size() > 0)
			{
				ContactID = SearchData.getJSON("CONTACT_ID");
				// ---- B1: Remove from specific list(s) ----
				if(ContactList.size() > 0)
				{
					response = netcore0.contacts_remove_from_List(contact_type,Netcore_end_point,netcore0__Netcore_API_key,ContactList,ContactID);
				}
				// ---- B2: No list specified - remove contact entirely ----
				else
				{
					if(contact_type == "identified")
					{
						response = netcore0.Identified_Contacts_remove_from_CM(Netcore_end_point,netcore0__Netcore_API_key,identityVal);
					}
					else
					{
						response = netcore0.Anon_contact_remove_from_CM(Netcore_end_point,netcore0__Netcore_API_key,ContactID);
					}
				}
			}
			else
			{
				// Contact wasn't found in Netcore
				response = searchResult;
			}
		}
		else
		{
			// Search API call failed / returned non-success
			response = searchResult;
		}
	}
}
catch (e)
{
	// Ensure response is a clean Map before attaching the error,
	// in case a prior assignment left it in a non-Map state.
	response = Map();
	response.put("error",e);
}

// ==========================================================================
// Function   : Get Data Transfer Attribute Progress Status
// Purpose    : Given a data_transfer_id (extracted from either query
//              params or request body), counts how many related
//              netcore0__Attribute_Request records are still "Pending"
//              and returns a derived status for the parent Data_Transfer.
// Input Vars : crmAPIRequest (map) - dataTransferId may arrive via EITHER:
//                  params.arguments (stringified JSON) - checked first, or
//                  body.arguments   (stringified JSON) - fallback
// Output     : response (map) - { "status":"success", "dataTransferStatus":
//              "Running"|"Creating Attributes", "pendingCount":<n> } on
//              success, or { "status":"error", "message"/"error":<msg> }
// ==========================================================================
response = Map();
try 
{
	dataTransId = null;
	// STEP 1: Extract payload from Form Data / Query Parameters
	paramsMap = crmAPIRequest.get("params");
	if(paramsMap != null && paramsMap.get("arguments") != null)
	{
		// Parse stringified JSON object from the 'arguments' key
		argsStr = paramsMap.get("arguments");
		argsMap = argsStr.toMap();
		dataTransId = argsMap.get("dataTransferId");
	}
	// STEP 2: Fallback extraction for Raw Body JSON payload (if not found in Form Data)
	if(dataTransId == null && crmAPIRequest.get("body") != null)
	{
		bodyMap = crmAPIRequest.get("body").toMap();
		if(bodyMap.get("arguments") != null)
		{
			argsMap = bodyMap.get("arguments").toMap();
			dataTransId = argsMap.get("dataTransferId");
		}
	}
	// STEP 3: Execute core business logic if dataTransferId exists
	if(dataTransId != null)
	{
		// Fetch child 'Attribute Request' records related to parent 'Data Transfer' record ID
		outputList = zoho.crm.getRelatedRecords("netcore0__Attribute_Request","netcore0__Data_Transfer",dataTransId.toLong());
		pendingCount = 0;
		// Iterate through related attribute records to count pending entries
		if(outputList != null && outputList.size() > 0)
		{
			for each  record in outputList
			{
				recMap = record.toMap();
				status = recMap.get("netcore0__Request_Status");
				// Check for 'Pending' status (prevents null-pointer exceptions)
				if(status != null && status.equals("Pending"))
				{
					pendingCount = pendingCount + 1;
				}
			}
		}
		// STEP 4: Determine overall Data Transfer Status based on pending count
		DataTransferStatus = "";
		if(pendingCount == 0)
		{
			DataTransferStatus = "Running";
		}
		else
		{
			DataTransferStatus = "Creating Attributes";
		}
		// Build success response payload
		response.put("status","success");
		response.put("dataTransferStatus",DataTransferStatus);
		response.put("pendingCount",pendingCount);
	}
	else
	{
		// Handle missing parameter error
		response.put("status","error");
		response.put("message","dataTransferId could not be extracted from request payload.");
	}
}
catch (e)
{
	// Catch execution exceptions and safely stringify error details
	response.put("status","error");
	response.put("error",e.toString());
}
// Return output response map to API consumer
return response;

								apiOp = "equals";
								if(op == "Is not")
								{
									coqlOp = "!=";
									apiOp = "not_equal";
								}
								else if(op == "Before")
								{
									coqlOp = "<";
									apiOp = "less_than";
								}
								else if(op == "After")
								{
									coqlOp = ">";
									apiOp = "greater_than";
								}
								if(val.toString().trim() == "")
								{
									clause = if(op == "Is not",fieldApi + " is not null",fieldApi + " is null");
									whereClauses.add(clause);
								}
								else
								{
									whereClauses.add(fieldApi + " " + coqlOp + " '" + val.toString() + "'");
									criteriaParts.add("(" + fieldApi + ":" + apiOp + ":" + val.toString() + ")");
								}
							}
						}
					}
					whereStatement = whereClauses.toString(" and ");
					finalCriteria = criteriaParts.toString("and");
					// ---- Total Count Calculation ----
					CountDatadynamic_map = Map();
					CountDatadynamic_map.put("modulename",netcore0__Which_Zoho_object);
					countApiUrlConnector = "netcore0.zohocrm0.get_total_recoards";
					if(finalCriteria != "")
					{
						countApiUrlConnector = "netcore0.zohocrm0.get_total_recoards_withcriteria0";
						CountDatadynamic_map.put("criteria",encodeUrl(finalCriteria));
					}
					countResult = zoho.crm.invokeConnector(countApiUrlConnector,CountDatadynamic_map);
					if(countResult.get("status_code") == 200)
					{
						CountRes = countResult.get("response");
						totalCount = ifNull(CountRes.get("count"),0).toLong();
						if(totalCount > 0)
						{
							// Contact Type Determination
							contact_type = "anonymous";
							if(is_identifier_primary == true && (netcoreCeIdentifier == "EMAIL" || netcoreCeIdentifier == "MOBILE" || netcoreCeIdentifier == "CUST_ID"))
							{
								contact_type = "identified";
							}
							crlf = hexToText("0D0A");
							// ---- Total COQL pages needed (e.g. 1,00,000 / 2,000 = 50 pages) ----
							totalPages = (totalCount / coqlPageSize).ceil().toLong();
							pageDummyList = "a".rightPad(totalPages).replaceAll(" ",",a").toList();
							recordsProcessed = 0;
							pageNumber = 0;
							netcoreBatchNumber = 0;
							// ---- CSV accumulator state — persists ACROSS pages until csvMaxRows is hit ----
							csvHeaderList = list();
							primaryHeader = netcoreCeIdentifier;
							csvHeaderList.add("\"" + primaryHeader + "\"");
							if(parsedFieldMappings != null && parsedFieldMappings.size() > 0)
							{
								for each  mapping in parsedFieldMappings
								{
									destField = mapping.get("destination_field");
									if(destField != null && destField != primaryHeader && !csvHeaderList.contains("\"" + destField + "\""))
									{
										csvHeaderList.add("\"" + destField + "\"");
									}
								}
							}
							csvHeaderRow = csvHeaderList.toString(",") + crlf;
							csvContent = csvHeaderRow;
							recordsInCurrentCsv = 0;
							// ============================================================================
							// PAGE LOOP — one iteration = one COQL fetch of up to 2000 records
							// (for-each + break workaround, since Deluge doesn't reliably support while)
							// ============================================================================
							for each  pageIter in pageDummyList
							{
								if(recordsProcessed >= totalCount)
								{
									break;
								}
								pageNumber = pageNumber + 1;
								coqlQuery = "select " + selectFields + " from " + netcore0__Which_Zoho_object;
								if(whereStatement != null && whereStatement.trim() != "")
								{
									coqlQuery = coqlQuery + " where " + whereStatement;
								}
								else
								{
									coqlQuery = coqlQuery + " where id is not null";
								}
								coqlQuery = coqlQuery + " limit " + recordsProcessed + "," + coqlPageSize;
								dynamic_map = Map();
								dynamic_map.put("query",coqlQuery);
								searchResult = zoho.crm.invokeConnector("netcore0.zohocrm0.coqlquesy",dynamic_map);
								pageFetchCount = 0;
								if(searchResult.get("status_code") == 200)
								{
									responseRecoardsdata = searchResult.get("response");
									ListData = ifNull(responseRecoardsdata.get("data"),list());
									if(ListData.size() > 0)
									{
										for each  items in ListData
										{
											rowValues = list();
											IdentityVal = ifNull(items.get(ZohoIdentifier),"");
											rowValues.add("\"" + IdentityVal.toString().replaceAll("\"","\"\"") + "\"");
											if(parsedFieldMappings != null && parsedFieldMappings.size() > 0)
											{
												for each  mapping in parsedFieldMappings
												{
													destField = mapping.get("destination_field");
													if(destField != primaryHeader)
													{
														src = mapping.get("source_field");
														fieldVal = ifNull(items.get(src),"");
														rowValues.add("\"" + fieldVal.toString().replaceAll("\"","\"\"") + "\"");
													}
												}
											}
											// ---- Append this row to the CSV currently being built ----
											csvContent = csvContent + rowValues.toString(",") + crlf;
											recordsInCurrentCsv = recordsInCurrentCsv + 1;
											recordsProcessed = recordsProcessed + 1;
											// ---- CSV reached its 50,000-row cap — send it now ----
											if(recordsInCurrentCsv >= csvMaxRows)
											{
												batchStartTime = netcore0.get_current_datetime();
												netcoreBatchNumber = netcoreBatchNumber + 1;
												sendResult = netcore0.send_csv_to_netcore(csvContent,netcoreBatchNumber,contact_type,Netcore_end_point,netcore0__Netcore_API_key,list_enable,ContactList,ContactListName);
												batchSummary = Map();
												batchSummary.put("batch_number",netcoreBatchNumber);
												batchSummary.put("records_in_batch",recordsInCurrentCsv);
												batchSummary.put("status",sendResult.get("status"));
												UpdateLogDataMap = Map();
												UpdateLogDataMap.put("netcore0__Data_Transfer_Id",data_transfer_id);
												UpdateLogDataMap.put("netcore0__Log_Type","Historic");
												batchCompletedTime = netcore0.get_current_datetime();
												if(sendResult.get("status") == "Pass")
												{
													UpdateLogDataMap.put("netcore0__Log_Status","Pass");
													UpdateLogDataMap.put("Name",sendResult.get("request_id"));
													UpdateLogDataMap.put("netcore0__Job_Id",sendResult.get("job_id"));
													UpdateLogDataMap.put("netcore0__Job_Status",sendResult.get("job_status"));
													UpdateLogDataMap.put("netcore0__Start_Time",batchStartTime);
													UpdateLogDataMap.put("netcore0__Completed_On",batchCompletedTime);
													batchSummary.put("job_id",sendResult.get("job_id"));
													batchSummary.put("job_status",sendResult.get("job_status"));
													successCount = successCount + 1;
													totalRecordsSent = totalRecordsSent + recordsInCurrentCsv;
												}
												else
												{
													UpdateLogDataMap.put("netcore0__Start_Time",batchStartTime);
													UpdateLogDataMap.put("netcore0__Completed_On",batchCompletedTime);
													UpdateLogDataMap.put("netcore0__Log_Status","Fail");
													UpdateLogDataMap.put("netcore0__Error_Message",sendResult.get("error"));
													UpdateLogDataMap.put("Name","Error");
													batchSummary.put("error",sendResult.get("error"));
													failCount = failCount + 1;
												}
												batchResults.add(batchSummary);
												CreateLog = zoho.crm.createRecord("netcore0__Data_Transfer_Logs",UpdateLogDataMap);
												// ---- Reset CSV accumulator for the next file ----
												csvContent = csvHeaderRow;
												recordsInCurrentCsv = 0;
											}
										}
										pageFetchCount = ListData.size();
									}
								}
								// Safety: if a page returns fewer than requested (or fails), stop paging further
								if(pageFetchCount < coqlPageSize)
								{
									recordsProcessed = totalCount;
									// forces outer loop to exit on next check
								}
							}
							// ============================================================================
							// Send any FINAL partial CSV that didn't reach csvMaxRows
							// (e.g. the remaining 50,000 in your 1,00,000-record example)
							// ============================================================================
							if(recordsInCurrentCsv > 0)
							{
								batchStartTime = netcore0.get_current_datetime();
								netcoreBatchNumber = netcoreBatchNumber + 1;
								sendResult = netcore0.send_csv_to_netcore(csvContent,netcoreBatchNumber,contact_type,Netcore_end_point,netcore0__Netcore_API_key,list_enable,ContactList,ContactListName);
								batchSummary = Map();
								batchSummary.put("batch_number",netcoreBatchNumber);
								batchSummary.put("records_in_batch",recordsInCurrentCsv);
								batchSummary.put("status",sendResult.get("status"));
								UpdateLogDataMap = Map();
								UpdateLogDataMap.put("netcore0__Data_Transfer_Id",data_transfer_id);
								UpdateLogDataMap.put("netcore0__Log_Type","Historic");
								batchCompletedTime = netcore0.get_current_datetime();
								if(sendResult.get("status") == "Pass")
								{
									UpdateLogDataMap.put("netcore0__Log_Status","Pass");
									UpdateLogDataMap.put("Name",sendResult.get("request_id"));
									UpdateLogDataMap.put("netcore0__Job_Id",sendResult.get("job_id"));
									UpdateLogDataMap.put("netcore0__Job_Status",sendResult.get("job_status"));
									batchSummary.put("job_id",sendResult.get("job_id"));
									batchSummary.put("job_status",sendResult.get("job_status"));
									UpdateLogDataMap.put("netcore0__Start_Time",batchStartTime);
									UpdateLogDataMap.put("netcore0__Completed_On",batchCompletedTime);
									successCount = successCount + 1;
									totalRecordsSent = totalRecordsSent + recordsInCurrentCsv;
								}
								else
								{
									UpdateLogDataMap.put("netcore0__Start_Time",batchStartTime);
									UpdateLogDataMap.put("netcore0__Completed_On",batchCompletedTime);
									UpdateLogDataMap.put("netcore0__Log_Status","Fail");
									UpdateLogDataMap.put("netcore0__Error_Message",sendResult.get("error"));
									UpdateLogDataMap.put("Name","Error");
									batchSummary.put("error",sendResult.get("error"));
									failCount = failCount + 1;
								}
								batchResults.add(batchSummary);
								CreateLog = zoho.crm.createRecord("netcore0__Data_Transfer_Logs",UpdateLogDataMap);
							}
							// ============================================================================
							// CONSOLIDATED RESPONSE AFTER ALL CSV BATCHES COMPLETE
							// ============================================================================
							response.put("status",if(failCount == 0,"success",if(successCount == 0,"failed","partial_success")));
							response.put("total_pages_fetched",pageNumber);
							response.put("total_csv_batches",netcoreBatchNumber);
							response.put("success_batches",successCount);
							response.put("failed_batches",failCount);
							response.put("total_records_sent",totalRecordsSent);
							response.put("batches",batchResults);
						}
						else
						{
							response.put("status","no_records");
							response.put("message","No matching records found to export.");
						}
					}
					else
					{
						response.put("status","failed");
						response.put("message","Failed to fetch total record count from Netcore.");
					}
				}
			}
		}
	}
}
catch (e)
{
	response.put("status","failed");
	response.put("code",400);
	response.put("error","Unexpected error while processing task details: " + e);
}
return response;

// ==========================================================================
// Function   : send_csv_to_netcore
// Namespace  : netcore0
// Purpose    : Sends a pre-built CSV string to Netcore's /v5/import/create
//              endpoint as a multipart upload. Called once per completed
//              CSV batch (either a full 50,000-row batch mid-loop, or the
//              final partial batch after all pages are fetched).
// Arguments  :
//   csvContent        (string)  - full CSV content including header row
//   batchNumber       (integer) - used only for the CSV filename
//   contact_type      (string)  - "identified" or "anonymous"
//   netcoreEndpoint   (string)  - base URL from netcore0__Connections
//   netcoreApiKey     (string)  - api-key from netcore0__Connections
//   listEnable        (boolean) - whether to attach audience_details
//   contactList       (string)  - comma-separated numeric list IDs
//   contactListName   (string)  - comma-separated list names
// Returns    : map - { status: "Pass"|"Fail", request_id, job_id,
//                       job_status, error }
// ==========================================================================
result = Map();
try 
{
	csvFile = csvContent.toFile("contacts_export_" + batchNumber + ".csv");
	csvFile.setParamName("file");
	filesList = list();
	filesList.add(csvFile);
	p1 = Map();
	p1.put("stringPart","true");
	p1.put("paramName","import_type");
	p1.put("content","contact");
	filesList.add(p1);
	p2 = Map();
	p2.put("stringPart","true");
	p2.put("paramName","contact_type");
	p2.put("content",contact_type);
	filesList.add(p2);
	p3 = Map();
	p3.put("stringPart","true");
	p3.put("paramName","field_delimiter");
	p3.put("content",",");
	filesList.add(p3);
	p4 = Map();
	p4.put("stringPart","true");
	p4.put("paramName","import_state");
	p4.put("content","publish");
	filesList.add(p4);
	p5 = Map();
	p5.put("stringPart","true");
	p5.put("paramName","autocorrect");
	p5.put("content","true");
	filesList.add(p5);
	p6 = Map();
	p6.put("stringPart","true");
	p6.put("paramName","ignore_unmapped_attributes");
	p6.put("content","false");
	filesList.add(p6);
	if(listEnable == true && contactList != "")
	{
		idListStr = "";
		for each  AId in contactList.toList(",")
		{
			if(AId.trim() != "")
			{
				idListStr = idListStr + AId.trim().toNumber() + ",";
			}
		}
		idListStr = idListStr.trim().replaceAll(",$","");
		nameListStr = "";
		for each  Aname in contactListName.toList(",")
		{
			if(Aname.trim() != "")
			{
				nameListStr = nameListStr + "\"" + Aname.trim() + "\",";
			}
		}
		nameListStr = nameListStr.trim().replaceAll(",$","");
		audienceJsonStr = "[{\"id\":[" + idListStr + "],\"name\":[" + nameListStr + "]}]";
		p8 = Map();
		p8.put("stringPart","true");
		p8.put("paramName","audience_details");
		p8.put("content",audienceJsonStr);
		filesList.add(p8);
	}
	headersMap = Map();
	headersMap.put("api-key",netcoreApiKey);
	apiResponse = invokeurl
	[
		url :netcoreEndpoint + "/v5/import/create"
		type :POST
		headers:headersMap
		files:filesList
	];
	apiResponseData = if(apiResponse != null && apiResponse.containsKey("data") && apiResponse.get("data") != null,apiResponse.get("data"),apiResponse);
	if(apiResponse.get("code") == 200)
	{
		jobDetails = apiResponseData.getJSON("job_details");
		jobId = jobDetails.get("job_id").toString();
		jobStatus = jobDetails.get("job_status");
		result.put("status","Pass");
		result.put("request_id",apiResponse.get("request_id"));
		result.put("job_id",jobId);
		result.put("job_status",jobStatus);
	}
	else
	{
		AppendError = "";
		errorflag = false;
		Responseerrors = if(apiResponse != null && apiResponse.containsKey("errors") && apiResponse.get("errors") != null,apiResponse.get("errors"),List());
		if(Responseerrors.size() > 0)
		{
			for each  items in Responseerrors
			{
				errorfields = items.get("field");
				errormsg = items.get("message");
				AppendError = AppendError + errorfields + " " + errormsg;
				errorflag = true;
			}
		}
		if(errorflag == false)
		{
			AppendError = if(apiResponseData.containsKey("description"),apiResponseData.get("description"),"Unknown API response format.");
		}
		result.put("status","Fail");
		result.put("request_id",apiResponse.get("request_id"));
		result.put("error",AppendError);
	}
}
catch (e)
{
	result.put("status","Fail");
	result.put("error","Netcore API call failed: " + e);
}
return result;
